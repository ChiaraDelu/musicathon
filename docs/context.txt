### ELEVEN LABS API
import { ElevenLabsClient, play } from '@elevenlabs/elevenlabs-js';

const elevenlabs = new ElevenLabsClient({
    apiKey: "YOUR_API_KEY", // Defaults to process.env.ELEVENLABS_API_KEY
});

const audio = await elevenlabs.textToSpeech.convert(
  'JBFqnCBsd6RMkjVDRZzb', // voice_id
  {
    text: 'The first move is what sets everything in motion.',
    modelId: 'eleven_multilingual_v2',
    outputFormat: 'mp3_44100_128', // output_format
  }
);

await play(audio);



### LALAL.AI API

# API v1 — authenticate with the X-License-Key header
# 1. Upload an audio file -> returns a source "id"
curl -X POST https://www.lalal.ai/api/v1/upload/ \
  -H "X-License-Key: YOUR_LALAL_KEY" \
  -H "Content-Disposition: attachment; filename=song.mp3" \
  --data-binary @song.mp3

# 2. Split it into stems -> returns a "task_id"
curl -X POST https://www.lalal.ai/api/v1/split/stem_separator/ \
  -H "X-License-Key: YOUR_LALAL_KEY" \
  -H "Content-Type: application/json" \
  -d '{"source_id":"SOURCE_ID","presets":{"stem":"vocals"}}'

# 3. Poll for the result download URLs
curl -X POST https://www.lalal.ai/api/v1/check/ \
  -H "X-License-Key: YOUR_LALAL_KEY" \
  -H "Content-Type: application/json" \
  -d '{"task_ids":["TASK_ID"]}'
